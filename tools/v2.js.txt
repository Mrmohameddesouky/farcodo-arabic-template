  function gadgets(sel) { return $$(sel + ' .widget'); }
  function yes(v) { return /^(yes|true|1|نعم)$/i.test((v || '').trim()); }
  function safeUrl(u) {
    u = (u || '').trim();
    return /^(https?:\/\/|\/|#|mailto:|tel:)/i.test(u) ? u : '';
  }
  function readCfg(w) {
    var c = $('.widget-content', w) || w, txt = '';
    try {
      var h = c.innerHTML.replace(/<br\s*\/?>/gi, '\n').replace(/<\/(p|div|li)>/gi, '\n');
      txt = new DOMParser().parseFromString(h, 'text/html').body.textContent || '';
    } catch (e) { txt = c.textContent || ''; }
    var o = {};
    txt.split(/\n/).forEach(function (l) {
      l = l.replace(/\u00a0/g, ' ').trim();
      if (!l || l.charAt(0) === '#') return;
      var m = l.match(/^([A-Za-z_][\w-]*)\s*[=:]\s*(.*)$/);
      if (m) o[m[1].toLowerCase()] = m[2].trim();
    });
    var t = $('.title', w) || $('h3', w);
    o._title = t ? t.textContent.trim() : '';
    return o;
  }

    });
  }

  var FCACHE = 300000, sid = 0;
  function feedC(path, n) {
    var k = 'fc-f:' + path + ':' + n;
    try { var s = sessionStorage.getItem(k); if (s) { var o = JSON.parse(s); if (Date.now() - o.t < FCACHE) return Promise.resolve(o.e); } } catch (e) {}
    return feed(path, n).then(function (j) {
      var e = ((j.feed && j.feed.entry) || []).slice(0, n).map(norm);
      try { sessionStorage.setItem(k, JSON.stringify({ t: Date.now(), e: e })); } catch (x) {}
      return e;
    });
  }

  function homeSections() {
    if (!$('.fc-hero') || !$('#home-sections')) return;
    var io = ('IntersectionObserver' in window) ? new IntersectionObserver(function (es) {
      es.forEach(function (x) { if (x.isIntersecting) { io.unobserve(x.target); x.target.__load(); } });
    }, { rootMargin: '300px' }) : null, first = true;
    gadgets('#home-sections').forEach(function (g) {
      var c = readCfg(g); if (yes(c.hidden)) return;
      var label = (c.label || '').trim(), title = c._title || c.title || label; if (!title) return;
      var n = Math.max(1, Math.min(12, parseInt(c.count, 10) || 4));
      var layout = /^(grid|list|compact)$/.test(c.layout) ? c.layout : 'grid';
      var path = label ? '/feeds/posts/default/-/' + encodeURIComponent(label) : '/feeds/posts/default';
      var s = el('section', 'fc-sec fc-sec-' + layout), hd = el('header', 'fc-sec-head'), tw = el('div'), hid = 'fc-s-' + (++sid);
      if (first) { s.id = 'fc-sections'; first = false; }
      var h = el('h2', 'fc-sec-t', title); h.id = hid; tw.appendChild(h);
      if (c.subtitle) tw.appendChild(el('p', 'fc-sec-sub', c.subtitle));
      hd.appendChild(tw);
      var mt = c.more === undefined ? T.all : c.more, mu = safeUrl(c.url) || (label ? '/search/label/' + encodeURIComponent(label) : '');
      if (mt && !/^(none|no|-)$/i.test(mt) && mu) {
        var m = el('a', 'fc-more', mt); m.href = mu; m.setAttribute('aria-label', mt + ': ' + title); hd.appendChild(m);
      }
      s.setAttribute('aria-labelledby', hid); s.appendChild(hd);
      var st = el('p', 'fc-state', T.loading); st.setAttribute('role', 'status');
      var body = layout === 'compact' ? el('ol', 'fc-clist') : el('div', layout === 'list' ? 'fc-grid fc-list' : 'fc-grid');
      if (layout === 'grid') body.setAttribute('data-n', String(n <= 4 ? n : (n % 4 === 0 ? 4 : 3)));
      s.appendChild(st); s.appendChild(body);
      s.__load = function () {
        st.textContent = T.loading; st.hidden = false;
        feedC(path, n).then(function (es) {
          if (!es.length) { if (yes(c.empty)) st.textContent = T.empty; else if (s.parentNode) s.parentNode.removeChild(s); return; }
          st.hidden = true; body.textContent = '';
          es.forEach(function (p) { body.appendChild(layout === 'compact' ? crow(p) : card(p)); });
        }).catch(function () {
          st.textContent = T.err; st.hidden = false;
          var r = el('button', 'btn', T.retry); r.type = 'button'; r.addEventListener('click', s.__load); st.appendChild(r);
        });
      };
      g.parentNode.insertBefore(s, g.nextSibling);
      if (io) io.observe(s); else s.__load();
    });
  }

  function iconKey(h) {
    h = (h || '').toLowerCase();
    if (/^mailto:/.test(h)) return 'email';
    if (/\/feeds\/|\.rss|\/rss/.test(h)) return 'rss';
    if (/(^|\/\/|\.)(facebook|fb)\.com/.test(h)) return 'facebook';
    if (/(^|\/\/|\.)(twitter|x)\.com/.test(h)) return 'x';
    if (/instagram\.com/.test(h)) return 'instagram';
    if (/youtube\.com|youtu\.be/.test(h)) return 'youtube';
    if (/linkedin\.com/.test(h)) return 'linkedin';
    if (/(^|\/\/)t\.me|telegram\./.test(h)) return 'telegram';
    if (/wa\.me|whatsapp\./.test(h)) return 'whatsapp';
    if (/tiktok\.com/.test(h)) return 'tiktok';
    return 'link';
  }
